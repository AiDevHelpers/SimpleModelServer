```csharp
using System;
using System.Collections.Generic;

public class OrderPricingService
{
    private readonly ITaxRateProvider _taxRateProvider;
    private readonly IDiscountRepository _discountRepository;

    public OrderPricingService(ITaxRateProvider taxRateProvider, IDiscountRepository discountRepository)
    {
        _taxRateProvider = taxRateProvider;
        _discountRepository = discountRepository;
    }

    public OrderTotal CalculateTotal(Order order)
    {
        decimal subtotal = CalculateSubtotal(order);
        decimal discountAmount = GetDiscountAmount(order);
        decimal taxAmount = CalculateTax(subtotal, order.Customer.Region);
        decimal total = subtotal - discountAmount + taxAmount;

        return new OrderTotal
        {
            Subtotal = subtotal,
            DiscountAmount = discountAmount,
            Tax = taxAmount,
            Total = total
        };
    }

    private decimal CalculateSubtotal(Order order)
    {
        decimal subtotal = 0;
        foreach (var lineItem in order.LineItems)
        {
            subtotal += lineItem.Quantity * lineItem.Price;
        }
        return subtotal;
    }

    private decimal GetDiscountAmount(Order order)
    {
        var discount = _discountRepository.GetActiveDiscount(order.Customer);
        return discount != null ? discount.Amount : 0;
    }

    private decimal CalculateTax(decimal subtotal, string region)
    {
        var taxRate = _taxRateProvider.GetTaxRate(region);
        return subtotal * taxRate;
    }
}
```
