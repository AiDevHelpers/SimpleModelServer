using System;
using System.Collections.Generic;

public class OrderPricingService
{
    private readonly ITaxRateProvider _taxRateProvider;
    private readonly IDiscountRepository _discountRepository;

    public OrderPricingService(ITaxRateProvider taxRateProvider, IDiscountRepository discountRepository)
    {
        _taxRateProvider = taxRateProvider ?? throw new ArgumentNullException(nameof(taxRateProvider));
        _discountRepository = discountRepository ?? throw new ArgumentNullException(nameof(discountRepository));
    }

    public OrderTotal CalculateTotal(Order order)
    {
        if (order == null) throw new ArgumentNullException(nameof(order));

        decimal subtotal = 0;
        foreach (var item in order.LineItems)
        {
            subtotal += item.Price * item.Quantity;
        }

        var discount = _discountRepository.GetActiveDiscount(order.CustomerId);
        decimal discountAmount = 0;
        if (discount != null)
        {
            discountAmount = subtotal * discount.DiscountPercentage / 100;
        }

        decimal taxRate = _taxRateProvider.GetTaxRate(order.CustomerRegion);
        decimal taxAmount = subtotal * taxRate;

        decimal total = subtotal - discountAmount + taxAmount;

        return new OrderTotal
        {
            Subtotal = subtotal,
            DiscountAmount = discountAmount,
            Tax = taxAmount,
            Total = total
        };
    }
}
