using System;
using System.Collections.Generic;
using System.Threading;

public class OrderPricingService
{
    private readonly ITaxRateProvider _taxRateProvider;
    private readonly IDiscountRepository _discountRepository;

    public OrderPricingService(ITaxRateProvider taxRateProvider, IDiscountRepository discountRepository)
    {
        _taxRateProvider = taxRateProvider ?? throw new ArgumentNullException(nameof(taxRateProvider));
        _discountRepository = discountRepository ?? throw new ArgumentNullException(nameof(discountRepository));
    }

    public OrderTotal CalculateTotal(Order order)
    {
        if (order == null) throw new ArgumentNullException(nameof(order));

        var subtotal = CalculateSubtotal(order);
        var discount = GetActiveDiscount(order);
        var tax = CalculateTax(order, discount);

        return new OrderTotal
        {
            Subtotal = subtotal,
            DiscountAmount = discount,
            Tax = tax,
            Total = subtotal - discount + tax
        };
    }

    private decimal CalculateSubtotal(Order order)
    {
        decimal subtotal = 0;
        foreach (var lineItem in order.LineItems)
        {
            subtotal += lineItem.Quantity * lineItem.UnitPrice;
        }
        return subtotal;
    }

    private decimal GetActiveDiscount(Order order)
    {
        var discounts = _discountRepository.GetActiveDiscounts(order.CustomerId);
        decimal discount = 0;
        foreach (var discount in discounts)
        {
            if (discount.AppliesTo(order))
            {
                discount = discount.Amount;
                break;
            }
        }
        return discount;
    }

    private decimal CalculateTax(Order order, decimal discount)
    {
        var taxRate = _taxRateProvider.GetTaxRate(order.CustomerRegion);
        return (CalculateSubtotal(order) - discount) * taxRate;
    }
}
