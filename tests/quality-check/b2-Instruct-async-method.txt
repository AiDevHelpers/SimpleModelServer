using System;
using System.Threading;
using System.Threading.Tasks;

public class InventoryService
{
    private readonly IStockRepository _stockRepository;
    private readonly InventoryReservation _reservation;
    private readonly CancellationToken _cancellationToken;

    public InventoryService(IStockRepository stockRepository, InventoryReservation reservation, CancellationToken cancellationToken)
    {
        _stockRepository = stockRepository;
        _reservation = reservation;
        _cancellationToken = cancellationToken;
    }

    public async Task<InventoryReservation> ReserveStockAsync(Guid productId, int quantity, CancellationToken cancellationToken)
    {
        cancellationToken.ThrowIfCancellationRequested();

        var currentStock = await _stockRepository.GetCurrentStockAsync(productId, cancellationToken);
        if (currentStock < quantity)
        {
            throw new InsufficientStockException("Not enough stock available.");
        }

        // Simulate transactional decrement of stock
        await _stockRepository.DecrementStockAsync(productId, quantity, cancellationToken);

        // Calculate expiration time 15 minutes from now
        var expiresAtUtc = DateTime.UtcNow.AddMinutes(15);

        // Create and return the reservation
        return new InventoryReservation(productId, expiresAtUtc);
    }
}
